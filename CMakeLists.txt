cmake_minimum_required(VERSION 3.16)
project(Milo VERSION 0.3.0 LANGUAGES CXX)
add_executable(milo native/milo.cpp native/bytecode.hpp)
target_compile_features(milo PRIVATE cxx_std_17)
set_target_properties(milo PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(milo PRIVATE /W4 /utf-8)
else()
  target_compile_options(milo PRIVATE -Wall -Wextra -Wpedantic)
endif()
